# Retroflag GPi Case 2 + Compute Module 4 Lite: firmware configuration (boot
# partition, p1). Read by the VideoCore firmware (start4.elf) before the ARM
# cores start. Same SoC and boot chain as the Raspberry Pi 4: see
# board/rpi4/config.txt for why most of these lines are what they are.
# board/bcm2711/post-build.sh checks the lines it depends on.
#
# Stage 1 (docs/boards/gpicase2.md): docked only. Video and sound on the dock's
# HDMI, pads and a USB Ethernet adapter on the dock's USB ports. The case's LCD
# (DPI on GPIO0-25 except 18) is not driven yet.

# --- What to boot ---------------------------------------------------------
arm_64bit=1
start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image.gz
device_tree=bcm2711-rpi-cm4.dtb
auto_initramfs=0

# --- Display: KMS through vc4 + v3d ---------------------------------------
# The dock's HDMI is the CM4's HDMI0 (HDMI-A-1); cmdline.txt switches HDMI-A-2
# off, as Retroflag's and Recalbox's setups do.
dtoverlay=vc4-kms-v3d-pi4,cma-256
max_framebuffers=2
disable_overscan=1
# The firmware leaves HDMI alone and vc4 alone turns it on (one TV lock per
# boot instead of two; board/rpi4/config.txt).
hdmi_ignore_hotplug=1

# --- Fast boot ------------------------------------------------------------
boot_delay=0
disable_splash=1
camera_auto_detect=0
display_auto_detect=0

# --- Radios: off (networking is a USB Ethernet adapter in the dock) ---------
# Harmless on a CM4 without Wi-Fi.
dtoverlay=disable-wifi
dtparam=krnbt=off

# --- USB ------------------------------------------------------------------
# The CM4's USB 2.0 port is an OTG port, off by default. Everything in the case
# hangs off it as a host: the dock's two ports, the built-in pad (a USB Xbox 360
# controller) and the case's USB sound card.
dtoverlay=dwc2,dr_mode=host

# --- The case's power switch and power latch ------------------------------
# Switch on GPIO26 (pull-up; sliding it off pulls it low): gpio-shutdown makes
# it a KEY_POWER that volumed turns into a clean poweroff.
dtoverlay=gpio-shutdown,gpio_pin=26
# Latch on GPIO27: the case stays powered while it is high. The firmware drives
# it high from the start; gpio-poweroff drives it low at the very end of a
# poweroff, after the filesystems are unmounted, and the case cuts the power.
gpio=27=op,dh
dtoverlay=gpio-poweroff,gpiopin=27,active_low=1

# --- CPU ------------------------------------------------------------------
arm_boost=1

# --- No serial console ----------------------------------------------------
# GPIO14/15 are the LCD's data lines in this case (DPI), and the cartridge
# gives no access to them anyway. Bring-up is over HDMI and SSH.
enable_uart=0
