# Raspberry Pi 4 firmware configuration (boot partition, p1). Read by the
# VideoCore firmware (start4.elf) before the ARM cores start.
# Reference: https://www.raspberrypi.com/documentation/computers/config_txt.html
# board/bcm2711/post-build.sh checks the lines it depends on.

# --- What to boot ---------------------------------------------------------
arm_64bit=1
start_file=start4.elf
fixup_file=fixup4.dat
# gzip-compressed (BR2_LINUX_KERNEL_IMAGEGZ): ~10 MB to read instead of the
# 29 MB Image; the firmware unpacks it. The first images booted the plain Image.
kernel=Image.gz
# Explicit, so the firmware never guesses (the built DTB is the only one on p1).
device_tree=bcm2711-rpi-4-b.dtb
# No initramfs: the root filesystem is mounted straight from the SD card.
auto_initramfs=0

# --- Display: KMS through vc4 + v3d ---------------------------------------
# The Pi 4 variant explicitly, rather than relying on overlay_map.dtb to map
# vc4-kms-v3d to it. cma-256: 256 MiB of CMA for the scanout buffers, as on
# the H616 boards (cma=256M there); 1080p double buffering plus Mesa's needs.
dtoverlay=vc4-kms-v3d-pi4,cma-256
max_framebuffers=2
disable_overscan=1
# The firmware leaves HDMI alone; the kernel's vc4 driver is the only thing
# that turns it on. Without this the TV locked TWICE on every boot: the
# firmware started HDMI (~5 s after power-on, a black screen since the splash
# is off), then vc4 took over at kernel 0.35 s with a full modeset (drm.debug:
# "modeset on [ENCODER:34:TMDS-34]"), restarting HDMI, and the TV searched for
# the signal again while the menu was already being drawn. The launcher adds no
# modeset of its own (same mode: "keeps [ENCODER...]"). With the firmware off
# HDMI: no simple-framebuffer, vc4 up at 0.17 s instead of 0.30 s, one lock
# (user-verified, 2026-10-19). vc4 detects the TV itself; the firmware's
# video= on the kernel command line goes away too, and the launcher picks the
# mode as it does on the H616 boards.
hdmi_ignore_hotplug=1

# --- Fast boot ------------------------------------------------------------
boot_delay=0
disable_splash=1
# Nothing to detect: no camera, no DSI display.
camera_auto_detect=0
display_auto_detect=0

# Onboard radios off for now (docs/boards/rpi4.md: USB pad first, onboard
# Bluetooth later). Without these, udev loaded brcmfmac + cfg80211 (no
# firmware for either: "Direct firmware load ... failed") and hci_uart at every
# boot, alongside rcS. krnbt=off only stops the kernel binding the Bluetooth
# chip (the UART stays as it is); remove both lines when the radios are wanted.
dtoverlay=disable-wifi
dtparam=krnbt=off

# --- CPU ------------------------------------------------------------------
# 1.8 GHz on boards that support it (newer Pi 4 revisions); ignored otherwise.
arm_boost=1

# --- Serial console -------------------------------------------------------
# GPIO14/15 (header pins 8 and 10), 3.3 V, 115200 8N1. With Bluetooth left on
# the PL011, the header UART is the mini UART, ttyS0 (serial0); enable_uart
# also locks the core clock so its baud rate stays right.
enable_uart=1
